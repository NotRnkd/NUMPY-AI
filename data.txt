def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return None
    return a / b

def greet(name):
    print("Hello, " + name + "!")

def is_even(number):
    if number % 2 == 0:
        return True
    return False

def count_to_n(n):
    for i in range(n):
        print(i)
